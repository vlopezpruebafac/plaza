-- ============================================================= -- PEDIDOS_PLAZA -- Pedidos que un mesero toma para un restaurante DISTINTO al dueño -- de la mesa donde está parado (pedidos "cruzados" dentro de la -- plaza de comidas). Ejecuta este script una sola vez en el -- SQL Editor de tu proyecto de Supabase, ANTES de usar la nueva -- versión de index.html. -- ============================================================= create table if not exists public.pedidos_plaza ( id uuid primary key default gen_random_uuid(), -- restaurante que debe preparar y cobrar este pedido restaurante_id uuid not null references public.restaurantes(id) on delete cascade, -- restaurante "dueño" de la mesa donde está el cliente (referencia informativa) restaurante_origen_id uuid references public.restaurantes(id) on delete set null, -- mesa y cliente donde se debe entregar el pedido (pueden quedar en null si se elimina la mesa) mesa_id uuid references public.mesas(id) on delete set null, cliente_id uuid references public.clientes(id) on delete set null, mesa_nombre text, cliente_nombre text, -- mesero que tomó el pedido mesero_id uuid references public.meseros(id) on delete set null, mesero_nombre text, -- productos pedidos: [{producto_id, nombre, precio, cantidad}, ...] items jsonb not null default '[]'::jsonb, subtotal numeric not null default 0, -- pendiente -> entregado_mesero -> cuenta_cobrada -> cerrado estado text not null default 'pendiente', creado_en timestamptz not null default now(), entregado_mesero_en timestamptz, cuenta_cobrada_en timestamptz, dinero_confirmado boolean not null default false, dinero_confirmado_en timestamptz ); create index if not exists idx_pedidos_plaza_restaurante on public.pedidos_plaza(restaurante_id); create index if not exists idx_pedidos_plaza_mesero on public.pedidos_plaza(mesero_id); create index if not exists idx_pedidos_plaza_mesa_rest_cliente on public.pedidos_plaza(mesa_id, restaurante_id, cliente_id); -- Habilita los eventos en tiempo real (realtime) para esta tabla, -- igual que ya están habilitados mesas / sillas / clientes / pedidos_items. alter publication supabase_realtime add table public.pedidos_plaza;
